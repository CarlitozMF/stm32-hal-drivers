55void Utils_DelayUs(uint32_t us)
56{
57 uint32_t cycles_per_us = SystemCoreClock / 1000000U;
58 if (cycles_per_us == 0U) cycles_per_us = 1U;
59
60#if UTILS_HAS_DWT
61 if (use_dwt) {
62 const uint32_t ticks = us * cycles_per_us;
63 const uint32_t start = DWT->CYCCNT;
64 while ((DWT->CYCCNT - start) < ticks) { }
65 return;
66 }
67#endif
68
69 /* ---- Respaldo: SysTick (cuenta hacia abajo y recarga desde LOAD) ---- */
70
71 /* Si SysTick no está corriendo, espera aproximada por software */
72 if ((SysTick->CTRL & SysTick_CTRL_ENABLE_Msk) == 0U) {
73 volatile uint32_t n = us * cycles_per_us / 4U + 1U;
74 while (n--) { }
75 return;
76 }
77
78 /* Si SysTick usa HCLK/8, cada tick dura 8 ciclos de CPU */
79 if ((SysTick->CTRL & SysTick_CTRL_CLKSOURCE_Msk) == 0U) {
80 cycles_per_us /= 8U;
81 if (cycles_per_us == 0U) cycles_per_us = 1U;
82 }
83
84 const uint32_t ticks = us * cycles_per_us;
85 const uint32_t reload = SysTick->LOAD + 1U;
86 uint32_t prev = SysTick->VAL;
87 uint32_t elapsed = 0U;
88
89 while (elapsed < ticks) {
90 uint32_t now = SysTick->VAL;
91 elapsed += (now <= prev) ? (prev - now) : (prev + reload - now);
92 prev = now;
93 }
94}
