52static void LCD_SendNibble(uint8_t nibble) {
53 HAL_GPIO_WritePin(_lcd.port_d4, _lcd.pin_d4, (nibble & 0x01) ? GPIO_PIN_SET : GPIO_PIN_RESET);
54 HAL_GPIO_WritePin(_lcd.port_d5, _lcd.pin_d5, (nibble & 0x02) ? GPIO_PIN_SET : GPIO_PIN_RESET);
55 HAL_GPIO_WritePin(_lcd.port_d6, _lcd.pin_d6, (nibble & 0x04) ? GPIO_PIN_SET : GPIO_PIN_RESET);
56 HAL_GPIO_WritePin(_lcd.port_d7, _lcd.pin_d7, (nibble & 0x08) ? GPIO_PIN_SET : GPIO_PIN_RESET);
57 LCD_PulseEnable();
58}

66static void LCD_SendByte(uint8_t byte, uint8_t is_data) {
67 // Configurar el pin Register Select (RS): 0 = Instrucción, 1 = Registro de datos
68 HAL_GPIO_WritePin(_lcd.port_rs, _lcd.pin_rs, is_data ? GPIO_PIN_SET : GPIO_PIN_RESET);
69
70 // Transmitir primero el nibble alto y luego el nibble bajo
71 LCD_SendNibble(byte >> 4);
72 LCD_SendNibble(byte & 0x0F);
73}

79static void LCD_PrintUInt(uint32_t value) {
80 char buffer[10]; // UINT32_MAX tiene 10 dígitos
81 uint8_t i = 0;
82
83 if (value == 0) {
84 LCD_WriteChar('0');
85 return;
86 }
87
88 // Extraer dígitos de menor a mayor peso
89 while (value > 0) {
90 buffer[i++] = (char)('0' + (value % 10));
91 value /= 10;
92 }
93
94 // Volcar los caracteres en orden correcto invirtiendo el búfer auxiliar
95 while (i > 0) {
96 LCD_WriteChar(buffer[--i]);
97 }
98}

104void LCD_Init(const LCD_Config_t *config) {
105 if (config == NULL) return;
106
107 _lcd = *config;
108 Utils_DelayUs_Init();
109
110 // Estado inicial seguro: todas las líneas en bajo (RS=0 es obligatorio para la secuencia)
111 HAL_GPIO_WritePin(_lcd.port_rs, _lcd.pin_rs, GPIO_PIN_RESET);
112 HAL_GPIO_WritePin(_lcd.port_en, _lcd.pin_en, GPIO_PIN_RESET);
113 HAL_GPIO_WritePin(_lcd.port_d4, _lcd.pin_d4, GPIO_PIN_RESET);
114 HAL_GPIO_WritePin(_lcd.port_d5, _lcd.pin_d5, GPIO_PIN_RESET);
115 HAL_GPIO_WritePin(_lcd.port_d6, _lcd.pin_d6, GPIO_PIN_RESET);
116 HAL_GPIO_WritePin(_lcd.port_d7, _lcd.pin_d7, GPIO_PIN_RESET);
117
118 HAL_Delay(50); // Estabilización inicial de VDD (>15 ms tras encendido)
119
120 // Secuencia obligatoria de reseteo por software para forzar el modo de 4 bits
121 LCD_SendNibble(0x03); HAL_Delay(5); // > 4.1 ms
122 LCD_SendNibble(0x03); HAL_Delay(1); // > 100 us
123 LCD_SendNibble(0x03); HAL_Delay(1);
124 LCD_SendNibble(0x02); HAL_Delay(1); // Conmutación definitiva a interfaz de 4 bits
125
126 // Comandos de configuración inicial (orden indicado por el datasheet)
127 LCD_SendByte(LCD_CMD_FUNCTION_SET, 0); // 4 bits, 2 líneas, matriz de 5x8 puntos
128 LCD_SendByte(LCD_CMD_DISPLAY_OFF, 0); // Display apagado durante la configuración
129 LCD_Clear(); // Borra la memoria y posiciona el cursor en (0,0)
130 LCD_SendByte(LCD_CMD_ENTRY_MODE, 0); // Cursor incrementa hacia la derecha, sin shift
131 LCD_SendByte(LCD_CURSOR_OFF, 0); // Display ENCENDIDO, cursor APAGADO, sin parpadeo
132}

139void LCD_SetCursor(uint8_t row, uint8_t col) {
140 // Direcciones base de DDRAM de cada fila. En el 20x4 las filas 2 y 3 son
141 // continuación de las filas 0 y 1 (por eso 0x14 y 0x54).
142 static const uint8_t row_offsets[4] = {0x00, 0x40, 0x14, 0x54};
143
144 uint8_t max_row = (_lcd.type == LCD_20X4) ? 3 : 1;
145 uint8_t max_col = (_lcd.type == LCD_20X4) ? 19 : 15;
146
147 // Saturar parámetros para evitar accesos fuera de la tabla o direcciones inválidas
148 if (row > max_row) row = max_row;
149 if (col > max_col) col = max_col;
150
151 LCD_SendByte((uint8_t)(LCD_CMD_SET_DDRAM + row_offsets[row] + col), 0);
152}

174void LCD_CreateCustomChar(uint8_t location, const uint8_t charmap[]) {
175 if (charmap == NULL) return;
176
177 location &= 0x7; // Asegurar rango válido de 0 a 7
178 LCD_SendByte((uint8_t)(LCD_CMD_SET_CGRAM | (location << 3)), 0); // Apuntar a la CGRAM
179 for (uint8_t i = 0; i < 8; i++) {
180 LCD_SendByte(charmap[i], 1);
181 }
182
183 // IMPORTANTE: tras escribir en CGRAM el contador de direcciones sigue apuntando a la CGRAM.
184 // Hay que volver a la DDRAM, de lo contrario los siguientes caracteres no se verían en pantalla.
185 LCD_SendByte(LCD_CMD_SET_DDRAM, 0);
186}

202void LCD_PrintFloat(float value, uint8_t decimal_places) {
203 // Casos especiales: NaN (único valor distinto de sí mismo)
204 if (value != value) {
205 LCD_Print("NaN");
206 return;
207 }
208
209 if (decimal_places > LCD_FLOAT_MAX_DECIMALS) {
210 decimal_places = LCD_FLOAT_MAX_DECIMALS;
211 }
212
213 uint8_t is_negative = (value < 0.0f);
214 if (is_negative) value = -value;
215
216 // Fuera de rango (incluye +/- infinito): evita overflow en la conversión a entero
217 if (value > LCD_FLOAT_MAX_VALUE) {
218 LCD_Print(is_negative ? "-Ovf" : "Ovf");
219 return;
220 }
221
222 // Factor 10^decimal_places
223 uint32_t multiplier = 1;
224 for (uint8_t i = 0; i < decimal_places; i++) {
225 multiplier *= 10;
226 }
227
228 // Se redondea el valor COMPLETO escalado. Redondear solo la parte fraccionaria
229 // fallaba con casos como 0.999 (2 decimales), donde el redondeo "se llevaba" un dígito.
230 uint64_t scaled = (uint64_t)(value * (float)multiplier + 0.5f);
231 uint32_t int_part = (uint32_t)(scaled / multiplier);
232 uint32_t frac_part = (uint32_t)(scaled % multiplier);
233
234 // Mostrar el signo solo si el resultado redondeado no es cero (evita "-0.00")
235 if (is_negative && scaled != 0) {
236 LCD_WriteChar('-');
237 }
238
239 LCD_PrintUInt(int_part);
240
241 if (decimal_places > 0) {
242 LCD_WriteChar('.');
243
244 // Imprimir cada decimal de mayor a menor peso; esto rellena con ceros a
245 // izquierda automáticamente (ej: frac=5, 2 decimales -> "05") sin búfer auxiliar.
246 for (uint32_t div = multiplier / 10; div > 0; div /= 10) {
247 LCD_WriteChar((char)('0' + ((frac_part / div) % 10)));
248 }
249 }
250}

252void LCD_DrawProgressBar(uint8_t row, uint8_t col, uint8_t width, uint8_t percentage) {
253 if (width < 2) return; // Se necesitan al menos los dos corchetes (evita underflow de uint8_t)
254 if (percentage > 100) percentage = 100;
255
256 LCD_SetCursor(row, col);
257
258 uint8_t inner_width = width - 2; // Espacio útil interno descontando los corchetes [ y ]
259 uint8_t filled_chars = (uint8_t)(((uint16_t)percentage * inner_width) / 100);
260
261 // Dibujar delimitador de apertura
262 LCD_WriteChar('[');
263
264 // Sección activa de la barra con caracteres de avance '>'
265 for (uint8_t i = 0; i < filled_chars; i++) {
266 LCD_WriteChar('>');
267 }
268
269 // Resto del ancho con espacios en blanco
270 for (uint8_t i = filled_chars; i < inner_width; i++) {
271 LCD_WriteChar(' ');
272 }
273
274 // Dibujar delimitador de cierre
275 LCD_WriteChar(']');
276}

278void LCD_Test(lcd_type_t type) {
279 LCD_Clear();
280
281 // 1. Test básico de impresión por filas
282 LCD_SetCursor(0, 0);
283 LCD_Print("--- LCD TEST ---");
284
285 LCD_SetCursor(1, 0);
286 LCD_Print("Driver OK");
287
288 if (type == LCD_20X4) {
289 LCD_SetCursor(2, 0);
290 LCD_Print("Fila 3: 20x4 Activo");
291 LCD_SetCursor(3, 0);
292 LCD_Print("Fila 4: Verificando");
293 }
294
295 HAL_Delay(3000); // Pausa estática para lectura inicial
296
297 // 2. Test de CGRAM (carácter personalizado en índice 0: un corazón)
298 const uint8_t custom_char[8] = {
299 0x00, 0x0A, 0x1F, 0x1F, 0x0E, 0x04, 0x00, 0x00
300 };
301 LCD_CreateCustomChar(0, custom_char);
302
303 LCD_Clear();
304 LCD_SetCursor(0, 0);
305 LCD_Print("Custom Char: ");
306 LCD_WriteChar(0); // Renderiza el carácter guardado en la posición 0 de la CGRAM
307
308 HAL_Delay(2000);
309
310 // 3. Test de números: enteros, flotantes y casos de borde
311 LCD_Clear();
312 LCD_SetCursor(0, 0);
313 LCD_PrintInt(-12345);
314 LCD_Print(" ");
315 LCD_PrintFloat(3.14159f, 2);
316 LCD_SetCursor(1, 0);
317 LCD_PrintFloat(0.05f, 2); // Verifica relleno de ceros ("0.05")
318 LCD_Print(" ");
319 LCD_PrintFloat(0.999f, 2); // Verifica acarreo del redondeo ("1.00")
320 HAL_Delay(3000);
321
322 // 4. Test de barra de progreso
323 LCD_Clear();
324 for (uint8_t p = 0; p <= 100; p += 10) {
325 LCD_DrawProgressBar(0, 0, 16, p);
326 LCD_SetCursor(1, 0);
327 LCD_PrintInt(p);
328 LCD_Print("% ");
329 HAL_Delay(250);
330 }
331
332 // 5. Test de modos del cursor (parpadeo)
333 LCD_Clear();
334 LCD_SetCursor(1, 0);
335 LCD_Print("Cursor Blink ON");
336 LCD_SetCursorMode(LCD_CURSOR_BLINK);
337 HAL_Delay(2500);
338 LCD_SetCursorMode(LCD_CURSOR_OFF); // Desactivar el cursor al finalizar la prueba
339
340 // 6. Test de desplazamiento dinámico de pantalla (Shift)
341 LCD_Clear();
342 LCD_SetCursor(0, 0);
343 LCD_Print("Desplazando...");
344 HAL_Delay(1000);
345
346 for (int i = 0; i < 3; i++) {
347 LCD_Shift(LCD_SHIFT_RIGHT);
348 HAL_Delay(400);
349 }
350 for (int i = 0; i < 3; i++) {
351 LCD_Shift(LCD_SHIFT_LEFT);
352 HAL_Delay(400);
353 }
354
355 // 7. Cierre de la secuencia de test (Clear también resetea el shift)
356 LCD_Clear();
357 LCD_SetCursor(0, 0);
358 LCD_Print("Test Finalizado!");
359}
